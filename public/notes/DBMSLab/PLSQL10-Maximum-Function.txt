-- CREATE FUNCTION
CREATE OR REPLACE FUNCTION findMax(x IN NUMBER, y IN NUMBER)
RETURN NUMBER
IS
    z NUMBER;
BEGIN
    IF x > y THEN
        z := x;
    ELSE
        z := y;
    END IF;

    RETURN z;
END;
/

-- DECLARE AND EXECUTE
DECLARE
    a NUMBER;
    b NUMBER;
    c NUMBER;
BEGIN
    a := 23;
    b := 45;

    c := findMax(a, b);

    DBMS_OUTPUT.PUT_LINE('Maximum of (23,45): ' || c);
END;
/
