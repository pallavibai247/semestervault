-- CREATE STUDENT TABLE
CREATE TABLE student (
    id NUMBER PRIMARY KEY,
    name VARCHAR2(30)
);

-- PACKAGE SPECIFICATION
CREATE OR REPLACE PACKAGE STUD_PKG AS
    PROCEDURE ADD_STUDENT(
        P_ID NUMBER,
        P_NAME VARCHAR2
    );
END STUD_PKG;
/

-- PACKAGE BODY
CREATE OR REPLACE PACKAGE BODY STUD_PKG AS
    PROCEDURE ADD_STUDENT(
        P_ID NUMBER,
        P_NAME VARCHAR2
    )
    IS
    BEGIN
        INSERT INTO student VALUES (P_ID, P_NAME);

        COMMIT;

        DBMS_OUTPUT.PUT_LINE('Student Added');
    END ADD_STUDENT;
END STUD_PKG;
/

-- EXECUTE PROCEDURE
BEGIN
    STUD_PKG.ADD_STUDENT(1, 'Ravi');
END;
/

-- DISPLAY STUDENT RECORDS
SELECT * FROM student;
