-- CREATE DEPARTMENT TABLE
CREATE TABLE department (
    dept_id NUMBER(5) PRIMARY KEY,
    dept_name VARCHAR2(50)
);

-- DISPLAY DEPARTMENT TABLE
DESC department;

-- INSERT DEPARTMENT RECORDS
INSERT INTO department VALUES (10, 'HR');
INSERT INTO department VALUES (20, 'IT');
INSERT INTO department VALUES (30, 'Sales');

-- CREATE EMPLOYEE TABLE
CREATE TABLE employee (
    emp_id NUMBER(5) PRIMARY KEY,
    emp_name VARCHAR2(50),
    salary NUMBER(10,2),
    dept_id NUMBER(5)
);

-- DISPLAY EMPLOYEE TABLE
DESC employee;

-- INSERT EMPLOYEE RECORDS
INSERT INTO employee VALUES (101, 'Anil', 50000, 10);
INSERT INTO employee VALUES (102, 'Ramesh', 60000, 20);
INSERT INTO employee VALUES (103, 'Suresh', 45000, 10);
INSERT INTO employee VALUES (104, 'Priya', 70000, 30);

-- DISPLAY EMPLOYEE RECORDS
SELECT * FROM employee;

-- CREATE VIEW
CREATE VIEW emp_view AS
SELECT emp_id, emp_name, salary
FROM employee;

-- DISPLAY VIEW
SELECT * FROM emp_view;

-- CREATE DEPARTMENT-WISE VIEW
CREATE VIEW dept_view AS
SELECT dept_id, COUNT(*) AS total_employees,
       AVG(salary) AS avg_salary
FROM employee
GROUP BY dept_id;

-- DISPLAY DEPARTMENT-WISE VIEW
SELECT * FROM dept_view;

-- UPDATE VIEW
UPDATE emp_view
SET salary = 65000
WHERE emp_id = 102;

-- DISPLAY UPDATED VIEW
SELECT * FROM emp_view;

-- DROP VIEW
DROP VIEW dept_view;
DROP VIEW emp_view;
