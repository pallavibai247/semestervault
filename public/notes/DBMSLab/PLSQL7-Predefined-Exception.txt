-- PROGRAM 7: HANDLING PREDEFINED EXCEPTION

DECLARE
    n NUMBER(4);
    d NUMBER(4);
    res NUMBER(4);

BEGIN
    DBMS_OUTPUT.PUT_LINE('Enter two numbers to divide');

    n := &n;
    d := &d;

    res := n / d;

EXCEPTION
    WHEN ZERO_DIVIDE THEN
        DBMS_OUTPUT.PUT_LINE('divide by error exception is caught');

    WHEN OTHERS THEN
        DBMS_OUTPUT.PUT_LINE('Result is ' || res);

END;
/
