DECLARE
    a NUMBER;
    b NUMBER;
    c NUMBER;

BEGIN
    a := &a;
    b := &b;
    c := &c;

    IF (a >= b AND a >= c) THEN
        DBMS_OUTPUT.PUT_LINE('A is largest ' || a);

    ELSIF (b >= a AND b >= c) THEN
        DBMS_OUTPUT.PUT_LINE('B is largest ' || b);

    ELSE
        DBMS_OUTPUT.PUT_LINE('C is the largest ' || c);
    END IF;
END;
/
