-- CREATE EMPLOYEE TABLE
CREATE TABLE employee (
    emp_id NUMBER(5) PRIMARY KEY,
    emp_name VARCHAR2(50),
    salary NUMBER(10,2),
    dept_id NUMBER
);

-- DISPLAY TABLE STRUCTURE
DESC employee;

-- INSERT RECORDS
INSERT INTO employee VALUES (101, 'Anil', 5000, 1);
INSERT INTO employee VALUES (102, 'Ramesh', 6000, 2);
INSERT INTO employee VALUES (103, 'Suresh', 7000, 3);

-- DISPLAY EMPLOYEE RECORDS
SELECT * FROM employee;

-- CREATE SAVEPOINT
SAVEPOINT SP1;

-- UPDATE EMPLOYEE SALARY
UPDATE employee
SET salary = 8000
WHERE emp_id = 101;

SELECT * FROM employee;

-- ROLLBACK TO SAVEPOINT
ROLLBACK TO SP1;

-- COMMIT CHANGES
COMMIT;

-- DISPLAY FINAL RECORDS
SELECT * FROM employee;
