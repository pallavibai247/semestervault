#!/bin/bash

read -p "Enter an integer: " n

n=${n#-}

if (( n == 0 )); then
    count=1
    sum=0
else
    count=0
    sum=0

    while (( n > 0 )); do
        sum=$((sum + n % 10))
        count=$((count + 1))
        n=$((n / 10))
    done
fi

echo "Number of digits = $count"
echo "Sum of digits = $sum"
