#!/bin/bash

read -p "Enter m: " m
read -p "Enter n (m<n): " n

echo "Prime numbers:"

for ((x=m; x<=n; x++)); do
    if (( x < 2 )); then
        continue
    fi

    prime=1

    for ((d=2; d*d<=x; d++)); do
        if (( x%d==0 )); then
            prime=0
            break
        fi
    done

    ((prime)) && printf "%d " "$x"
done

echo
